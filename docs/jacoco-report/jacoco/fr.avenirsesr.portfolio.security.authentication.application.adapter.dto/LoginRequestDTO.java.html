@Data
@Accessors(chain = true)
public class LoginRequestDTO {

  private String login;

  @ToString.Exclude private String password;

  @ToString.Include(name = "password")
  @Schema(hidden = true)
  public String getMaskedPassword() {
    return password != null ? "****" : null;
  }
}
